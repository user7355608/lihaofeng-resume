**$STP协议 如果校验和后又'N'、‘C’两个字节 地面站需要增加CRC16校验**

校验方法:
```C
uint32_t UtilityCrc32(uint32_t init, const uint32_t poly, const uint8_t* data, uint32_t len)
{
    const uint32_t polyTable[2] = {0, poly};
    uint32_t tmp = 0;
    for( ; len >=4; data += 4, len -=4)
    {
        memcpy(&tmp, data, 4);
        init^= tmp;
        for(uint8_t i = 0; i < 32; i++)
        {
            init = (init << 1) ^ polyTable[init >> 31];
        }
    }

    if(len)
    {
        tmp = 0;
        for(uint8_t i = 0; i < len; i++)
        {
            tmp |= (*data++) << (8 * i);
        }
        init^= tmp;
        for(uint8_t i = 0; i < 32; i++)
        {
            init = (init << 1) ^ polyTable[init >> 31];
        }
    }
    return init;
}
uint32_t  crc_ret = UtilityCrc32(0xFFFFFFFF, 0x04C11DB7, buf, len);

```

## 航线编辑  可串口测试 

飞控上传整条航线的逻辑是，先一个一个单点上传所有的航点，最后再发送航点的总数。期间飞控会回复地面站每个单点上传的点。
飞控接收到的航点总数与最后一条的航点总数相等，即认为航线上传完成。

/**
 * @fn static u16 CRC16_2(u8* pchMsg, u16 wDataLen)
 * @brief 计算指定数组的 CRC_16 结果
 * @param[in] pchMsg 待计算的数组指针
 * @param[in] wDataLen 数组长度
 * @return wCRC 计算完的校验
 */
uint16_t CRC16_2(const uint8_t* pchMsg, const uint32_t wDataLen) //高位在前
{
    u16 wCRC = 0xFFFF;
    u8 chChar;
    for (uint32_t i = 0; i < wDataLen; ++i)
    {
        chChar = *pchMsg++;
        wCRC   = wCRCTalbeAbs[(chChar ^ wCRC) & 15] ^ (wCRC >> 4);
        wCRC   = wCRCTalbeAbs[((chChar >> 4) ^ wCRC) & 15] ^ (wCRC >> 4);
    }
    return wCRC;
}
